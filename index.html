Country Controller

controllers/CountryController.php

<?php
require_once __DIR__ . '/../models/Country.php';

class CountryController
{
    private Country $model;

    public function __construct()
    {
        $this->model = new Country();
    }

    public function dashboard(): void
    {
        $keyword = trim($_GET['q'] ?? '');

        $countries = $keyword !== '' ? $this->model->search($keyword) : $this->model->getAll();
        $totalCountries  = $this->model->countCountries();
        $totalContinents = $this->model->countContinents();
        $totalPopulation = $this->model->getTotalPopulation();
        $mostPopulated   = $this->model->getMostPopulated();

        require __DIR__ . '/../views/dashboard.php';
    }
}

Country Model

models/Country.php

<?php
require_once __DIR__ . '/../../config/database.php';

class Country
{
    private PDO $pdo;

    public function __construct()
    {
        $this->pdo = (new Database())->connect();
    }

    public function getAll(): array
    {
        return $this->pdo->query('SELECT * FROM country ORDER BY name ASC')->fetchAll();
    }

    public function search(string $keyword): array
    {
        $st = $this->pdo->prepare('SELECT * FROM country WHERE name LIKE :k ORDER BY name ASC');
        $st->execute(['k' => '%' . $keyword . '%']);
        return $st->fetchAll();
    }

    public function countCountries(): int
    {
        return (int) $this->pdo->query('SELECT COUNT(*) FROM country')->fetchColumn();
    }

    public function countContinents(): int
    {
        return (int) $this->pdo->query('SELECT COUNT(DISTINCT continent) FROM country')->fetchColumn();
    }

    public function getTotalPopulation(): int
    {
        return (int) $this->pdo->query('SELECT SUM(population) FROM country')->fetchColumn();
    }

    public function getMostPopulated(): array
    {
        return $this->pdo->query('SELECT name, population FROM country ORDER BY population DESC LIMIT 1')->fetch()
            ?: ['name' => '-', 'population' => 0];
    }
}

Database Connection

config/database.php

<?php
class Database
{
    public function connect(): PDO
    {
        try {
            return new PDO('mysql:host=localhost;dbname=world_explorer;charset=utf8mb4','root','root');
        } catch (PDOException $e) {
            die('Erreur de connexion : ' . $e->getMessage());
        }
    }
}
