Country Model

models/Country.php

<?php
require_once __DIR__ . '/../../config/database.php';

class Country
{
    private PDO $pdo;

    public function __construct()
    {
        $this->pdo = (new Database())->connect();
    }

    public function getAll(): array
    {
        return $this->pdo->query('SELECT * FROM country ORDER BY name ASC')->fetchAll();
    }

    public function search(string $keyword): array
    {
        $st = $this->pdo->prepare('SELECT * FROM country WHERE name LIKE :k ORDER BY name ASC');
        $st->execute(['k' => '%' . $keyword . '%']);
        return $st->fetchAll();
    }

    public function countCountries(): int
    {
        return (int) $this->pdo->query('SELECT COUNT(*) FROM country')->fetchColumn();
    }

    public function countContinents(): int
    {
        return (int) $this->pdo->query('SELECT COUNT(DISTINCT continent) FROM country')->fetchColumn();
    }

    public function getTotalPopulation(): int
    {
        return (int) $this->pdo->query('SELECT SUM(population) FROM country')->fetchColumn();
    }

    public function getMostPopulated(): array
    {
        return $this->pdo->query('SELECT name, population FROM country ORDER BY population DESC LIMIT 1')->fetch()
            ?: ['name' => '-', 'population' => 0];
    }
}

require_once __DIR__ . "/../../app/models/Country.php";
class CountryController{
    private Country $country;

    public function __construct() {
        $this->country = new Country();
    }
    public function dashboard(){
        $name = trim($_GET['country'] ?? '');
        $countries = $name ? $this->country->search($name) : $this->country->getAll();
        $totalCountries = $this->country->countCountries();
        $totalContinents = $this->country->countContinents();
        $totalPopulation = $this->country->getTotalPopulation();
        $mostPopulated = $this->country->getMostPopulated();
        require_once __DIR__ . "/../../app/views/dashboard.php";
    }
    public function create(){
        require_once __DIR__ . "/../../app/views/create.php";
    }

    public function store(){
        $name = $_POST['name'];
        $iso_code = $_POST['iso_code'];
        $capital = $_POST['capital'];
        $continent = $_POST['continent'];
        $population = $_POST['population'];
        $currency = $_POST['currency'];

        $this->country->add($name, $iso_code, $capital, $continent, $population, $currency);

        header("Location: index.php");
        exit;
    }
    public function delete(){
        $id = $_GET['id'];

        $this->country->delete($id);

        header("Location: index.php");
        exit;
    }
}

require_once __DIR__ . "/../../config/connection.php";

class Country {
    private $pdo;
    public function __construct() {
        $this->pdo = DataBase::connection();
    }
    public function getAll(){
        return $this->pdo->query("SELECT * FROM country ORDER BY name ASC")->fetchAll();
    }
    public function search($name){
        return $this->pdo->query("SELECT * FROM country WHERE name like '%$name%'")->fetchAll();
    }
    public function countCountries(){
        return $this->pdo->query("SELECT COUNT(*) FROM country")->fetchColumn();
    }
    public function countContinents(){
        return $this->pdo->query("SELECT COUNT(DISTINCT continent) FROM country")->fetchColumn();
    }
    public function getTotalPopulation(){
        return $this->pdo->query("SELECT SUM(population) FROM country")->fetchColumn();
    }
    public function getMostPopulated(){
        return $this->pdo->query("SELECT name, population FROM country ORDER BY population DESC LIMIT 1")->fetch();
    }
    public function add($name, $iso_code, $capital, $continent, $population, $currency){
        return $this->pdo->query(
            "INSERT INTO country (name, iso_code, capital, continent, population, currency)
            VALUES ('$name', '$iso_code', '$capital', '$continent', '$population', '$currency')"
        );
    }
    public function delete($id){
        return $this->pdo->query("DELETE FROM country WHERE id = $id");
    }
}

Résumé — Fichiers, Upload, Sessions et Cookies

1. Manipulation des fichiers

FonctionRôle

fgets()Lire une ligne

Modes : r = lecture · w = écriture (efface l'ancien contenu) · a = ajout à la fin · r+ = lecture et écriture sans effacer.

2. Upload de fichiers

Formulaire obligatoire : method="POST" et enctype="multipart/form-data".

<form method="POST" enctype="multipart/form-data">
    <input type="file" name="fichier">
    <button type="submit">Envoyer</button>
</form>

$_FILES : name = nom · type = type MIME · size = taille · tmp_name = emplacement temporaire · error = code d'erreur.

Fonctions : pathinfo() récupère l'extension · strtolower() convertit en minuscules · in_array() vérifie une valeur · move_uploaded_file() déplace le fichier.

3. Gestion des sessions

Une session conserve les informations d'un utilisateur entre plusieurs requêtes HTTP. Les données sont stockées côté serveur.

session_start();
$_SESSION['user'] = "admin"; // Créer
echo $_SESSION['user'];      // Lire
unset($_SESSION['user']);   // Supprimer une variable
session_destroy();          // Détruire la session

Protéger une page :

session_start();
if (!isset($_SESSION['user'])) {
    header("Location: connexion.php");
    exit;
}

session_regenerate_id(true) renouvelle l'identifiant de session.

4. Gestion des cookies

Un cookie conserve une petite information côté navigateur (client).

setcookie('theme', 'dark', time() + 3600, '/'); // Créer
echo $_COOKIE['theme'];                         // Lire
setcookie('theme', '', time() - 3600, '/');    // Supprimer

Paramètres : expires = expiration · path = chemin · secure = HTTPS uniquement · httponly = inaccessible via JavaScript · samesite = contrôle l'envoi intersites.

5. Différences et rappels essentiels

SessionsCookies

Adaptées à l'authentificationAdaptés aux préférences

  • isset() vérifie si une variable existe ; header() envoie un en-tête HTTP ou effectue une redirection ; time() retourne le timestamp actuel.
